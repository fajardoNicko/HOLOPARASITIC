# Create phylogenetic tree ###########################
# using V.PhyloMaker 2 Jin and Qian 2022 - https://www.sciencedirect.com/science/article/pii/S2468265922000580

# installing PhyloMaker 2 (you may need to install package "ape" first)

# import species list
tx122<- read.csv("DataS1/UCBG_122spp_list.csv")
# create the phylogenetic tree
tree122 <- phylo.maker(sp.list = tx122%>%select (spp_name_WP, genus_WP, family_WP), tree = GBOTB.extended.WP, nodes = nodes.info.1.WP, scenarios = "S3")
# check if all 122 species were included
tree122$scenario.3$tip.label 
# write the tree
write.tree(tree122$scenario.3, "UCBG_122spp.tre")


# remove _ from phylogenetic tree tip labels
tree122$tip.label<- gsub('_',' ',tree122$tip.label)

# plot phylogenetic tree  
asterids<-rep("#D55E00", 30)
rosids<-rep("#CC79A7",40)
b_eudico<- rep("#F0E442",10) 
mono<-rep("#009E73",17) 
b_angio<- rep("#56B4E9", 13)
ferns<- rep("#999999", 12)
my_color<-c(asterids, rosids, b_eudico, mono, b_angio, ferns)
length (my_color)

pdf(width=15,height=15,file='Phylo_tree.pdf')
plot(x=tree122,tip.color=my_color, type='fan')
for (clade in clades_unique)
{
  arc.cladelabels(tree=tree122,text=clade,node=getMRCA(phy=tree122, tip=tx122$spp_name_WP[tx122$clade==clade]), ln.offset=10,
                  lab.offset=10)
  nodelabels(text=clade,node=getMRCA(phy=tree122, tip=tx122$spp_name_WP[tx122$clade==clade]))
}
dev.off()


# summarize taxa data
sum_tx122<- as.data.frame(tx122%>% 
  group_by (clade)%>%
  summarize (number_of_species = n()))

# reorder clade levels
sum_tx122$clade <- factor(sum_tx122$clade, 
                          levels = c("ferns", "basal angiosperms", "monocots",
                                     "basal eudicots", "rosids", "asterids"))

g_bar<-ggplot(data =sum_tx122, aes(x= clade, y = number_of_species, fill = clade))+

# Obtain species occurrences coordinates ################
# Use BIEN function to obtain coordinates and plot map of occurrence -----------
vec_names<-unique(tx122$spp_name_WP)
coord<- list()
for (i in 1:length(vec_names)) {
  # extract coordinates for species occurrence
  coord[[i]]<-BIEN_occurrence_species(vec_names[i]) 
  #plot species occurrence
  png(file=paste0(vec_names[i],"_plot.png"),
      width=600, height=350)
  map('world', fill = TRUE, col= "grey", bg = "light blue")
  points(cbind(coord[[i]]$longitude,
               coord[[i]]$latitude),
         col = "red",
         pch = 20,
         cex = 1) 
  print(vec_names[i])
  dev.off()
}
